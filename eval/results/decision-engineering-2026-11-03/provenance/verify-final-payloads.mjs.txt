// Offline compiler-only check; run from repository root with node --import=tsx --input-type=module < THIS_FILE.
import {readFile,writeFile} from 'node:fs/promises';
import {gunzipSync} from 'node:zlib';
import {createHash} from 'node:crypto';
const base='eval/results/decision-engineering-2026-11-03';
const hash=b=>createHash('sha256').update(b).digest('hex');
const freezeBytes=await readFile(base+'/source-freeze-final-reviewed.json');
const frozen=JSON.parse(freezeBytes);
for(const [file,expected]of Object.entries(frozen.files))if(hash(await readFile(file))!==expected)throw new Error('Final source mismatch');
const {compileCheck}=await import('./src/kernel/index.ts');
const {makeRequest,presentationOrder}=await import('./eval/backends.ts');
const checks=[];
for(const profile of ['balanced-v1','semantic-v2'])for(const dataset of ['vitaminc-combined-dev','scifact-dev','contract-diagnostic','batch20-original','batch20-reversed']){
 const report=JSON.parse(gunzipSync(await readFile(`${base}/runs/${profile}/${dataset}.report.json.gz`)));
 const batches=dataset.startsWith('batch20')?[report.rows]:report.rows.map(row=>[row]);
 let matched=0;
 for(const rows of batches){const pairs=rows.map(row=>row.pair);const request=makeRequest(pairs[0],pairs,profile);const compiled=compileCheck(request,'jev-1.14.0',{presentationOrder:presentationOrder(request,report.protocol.presentation,1)});const actual=hash(JSON.stringify(compiled.payload));for(const row of rows){if(row.prediction.prompt_hash!==actual)throw new Error('Payload identity mismatch');matched++;}}
 checks.push({profile,dataset,matched_pair_observations:matched});
}
for(const [file,expected]of Object.entries(frozen.files))if(hash(await readFile(file))!==expected)throw new Error('Source changed while checking');
const result={kind:'independent-offline-final-compiler-payload-check',final_source_freeze_sha256:hash(freezeBytes),model_calls:0,matched_pair_observations:checks.reduce((n,x)=>n+x.matched_pair_observations,0),checks};
await writeFile(base+'/provenance/final-payload-identity.json',JSON.stringify(result,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({offline_matched_pair_observations:result.matched_pair_observations,model_calls:0}));
